        graph TB
            %% Middleware Central atuando como ponte
            ROS2{ROS 2 Humble Middleware}

            %% Ambiente Físico (Caixa Sólida)
            subgraph Fisico [Ambiente Físico]
                direction TB
                RPi[Raspberry Pi 4]
                Sensor[Sensor IR BFD-1000]
                Roomba[Atuadores iRobot Roomba]
                
                Sensor -->|Leitura GPIO| RPi
                RPi -->|Comandos de Velocidade| Roomba
            end

            %% Ambiente Simulado (Caixa Tracejada)
            subgraph Simulado [Ambiente Simulado]
                direction TB
                Turtlesim[Turtlesim]
                RViz[RViz2]
                Aruco[Leitor de ArUco]
                
                Aruco -->|Posição Virtual| RViz
                Turtlesim -.->|Odometria/Lógica| RViz
            end

            %% Conexões do Middleware com os Ambientes
            ROS2 <==> RPi
            ROS2 <==> Turtlesim
            ROS2 <==> RViz

            style Simulado stroke:#333,stroke-width:2px,stroke-dasharray: 5 5,fill:#f8f9fa
